# M1 技術驗證 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 以實際環境驗證 LiteLLM 開源版（LLM + MCP gateway）、Keycloak token exchange 與 OBO 能否作為 AI Governance Gateway 的 data plane，產出 `docs/spikes/m1-findings.md` 決定 M2 架構。

**Architecture:** `docker compose` 啟動 LiteLLM、Postgres、Keycloak 與自寫的 `probe_mcp`（回傳收到的 header 與 token claims）。kyt_agent 以最小改動改走 LiteLLM。每個驗證項目（V1–V8）以腳本執行，結果記錄到 findings。

**Tech Stack:** LiteLLM v1.104.4、Keycloak 26.8.1、Postgres 17、Python 3.12、`mcp` SDK 1.x（FastMCP）、httpx、python-dotenv、pytest、uv。

**Spec:** `docs/specs/2026-10-20-m1-spike-design.md`

**通則**
- 這是技術驗證：設定若與實際版本行為不同，以實測為準，修正後把差異寫進 findings 的「與文件不符之處」
- 任何指令都不得輸出 token、API key、client secret 原文；只輸出 claims 或 `<redacted>`
- 版本已於 2026-10-20 查詢：LiteLLM 最新穩定版 v1.104.4、Keycloak 26.8.1、`mcp` 1.30.0（2.x 的 API 有變動，M1 固定用 1.x）

```python
import base64
import json

from probe_mcp.claims import decode_claims, summarize_headers


def _jwt(payload: dict) -> str:
    def part(data: dict) -> str:
        raw = json.dumps(data).encode()
        return base64.urlsafe_b64encode(raw).rstrip(b"=").decode()

    return f"{part({'alg': 'none'})}.{part(payload)}.sig"


def test_decode_claims_handles_missing_padding():
    assert decode_claims(_jwt({"sub": "alice", "aud": "probe-mcp"})) == {

        "aud": "probe-mcp",

def test_summarize_headers_decodes_bearer_and_redacts_secrets():
```

- [ ] **Step 3: 實作**

```python
import base64
import json
from collections.abc import Mapping
from typing import Any

# 只有這些 header 原樣顯示；其餘可能含憑證，一律只顯示長度
VISIBLE_HEADERS = {
    "host",
    "user-agent",
    "accept",
    "content-type",
    "content-length",
    "mcp-protocol-version",
    "mcp-session-id",
    "x-litellm-end-user-id",
    "x-litellm-key-alias",
    "x-litellm-team-id",
    "x-forwarded-for",
}


def decode_claims(token: str) -> dict[str, Any]:
    parts = token.split(".")
    if len(parts) != 3:
        return {"error": "not a JWT"}
    padded = parts[1] + "=" * (-len(parts[1]) % 4)
    try:
        claims: dict[str, Any] = json.loads(base64.urlsafe_b64decode(padded))
    except (ValueError, json.JSONDecodeError):
        return {"error": "invalid payload"}
    return claims


def summarize_headers(headers: Mapping[str, str]) -> dict[str, Any]:
    shown: dict[str, str] = {}
    claims: dict[str, Any] | None = None
    for name, value in headers.items():
        key = name.lower()
        if key == "authorization" and value.lower().startswith("bearer "):
            claims = decode_claims(value[7:])
            shown[key] = "<bearer>"
        elif key in VISIBLE_HEADERS:
            shown[key] = value
        else:
            shown[key] = f"<redacted len={len(value)}>"
    return {"headers": shown, "bearer_claims": claims}
```

```python
@mcp.tool(description=WHOAMI_DESCRIPTION)
def whoami(ctx: Context) -> dict:
    request = ctx.request_context.request

@mcp.tool(description="Echo the given text.")

    return text
```

- [ ] **Step 2: `scripts/kc.py`**

```python
import httpx

from scripts.env import KEYCLOAK_TOKEN_URL, env, user_password


def user_token(username: str, client_id: str = "kyt-cli") -> str:
    """Spike 用 password grant 取代 device flow，測試帳號密碼來自 realm 檔。"""
    response = httpx.post(
        KEYCLOAK_TOKEN_URL,
        data={
            "grant_type": "password",
            "client_id": client_id,
            "username": username,
            "password": user_password(username),
            "scope": "openid",
        },
    )
    response.raise_for_status()
    return str(response.json()["access_token"])


def client_token(client_id: str, secret_env: str) -> str:
    response = httpx.post(
        KEYCLOAK_TOKEN_URL,
        data={
            "grant_type": "client_credentials",
            "client_id": client_id,
            "client_secret": env(secret_env),
        },
    )
    response.raise_for_status()
    return str(response.json()["access_token"])
```

```python
async def run(key_alias: str, user: str | None) -> None:
    headers = {"x-litellm-api-key": f"Bearer {json.loads(KEYS_FILE.read_text())[key_alias]}"}
    if user:
        headers["Authorization"] = f"Bearer {user_token(user)}"

            tools = [tool.name for tool in (await session.list_tools()).tools]

            print(result.content[0].text if result.content else "<empty>")


if __name__ == "__main__":
```

```python
from typing import Any

from litellm.integrations.custom_guardrail import CustomGuardrail


class GovernanceHook(CustomGuardrail):
    """V7 驗證用：遇到 echo 且 text 含 freeze 時阻擋，觀察 hook 能拿到的資訊。"""

    async def async_pre_call_hook(
        self, user_api_key_dict: Any, cache: Any, data: dict, call_type: str
    ) -> dict:
        print(
            "governance_hook",
            call_type,
            getattr(user_api_key_dict, "key_alias", None),
            sorted(data.keys()),
            flush=True,
        )
        arguments = data.get("arguments") or {}
        if str(data.get("name", "")).endswith("echo") and "freeze" in str(arguments.get("text", "")):
            raise ValueError("pending_approval: approval_id=demo-123")
        return data
```
